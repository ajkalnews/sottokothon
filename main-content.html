async function fetchFeaturedNews() { try { // ১. মেইন লিড নিউজ নিয়ে আসা const leadRes = await fetch('/api/posts?type=lead'); const leadPosts = await leadRes.json(); if (leadPosts.length > 0) { const lead = leadPosts[0]; const leadHtml = `
${lead.image_caption || ''}

${lead.title}

${lead.content.substring(0, 150)}...

${timeAgo(lead.created_at)} | ক্যাটাগরি: ${lead.category_name || 'অন্যান্য'}
`; document.querySelector('.mi-featured').innerHTML = leadHtml; } // ২. ২টি সাব-নিউজ নিয়ে আসা const subRes = await fetch('/api/posts?type=sub'); const subPosts = await subRes.json(); if (subPosts.length > 0) { let subHtml = '
'; subPosts.forEach(post => { subHtml += `

${post.title}

${post.content.substring(0, 100)}...

${timeAgo(post.created_at)}
`; }); subHtml += '
'; // মেইন লিডের নিচে সাব-গ্রিড অ্যাপেন্ড করা document.querySelector('.mi-featured').innerHTML += subHtml; } } catch (error) { console.error("নিউজ লোড করতে ব্যর্থ:", error); } } // কত সময় আগে পোস্ট করা হয়েছে তা বের করার সিম্পল ফাংশন function timeAgo(dateString) { const now = new Date(); const postDate = new Date(dateString); const diffMs = now - postDate; const diffMins = Math.floor(diffMs / 60000); const diffHours = Math.floor(diffMins / 60); if (diffMins < 60) return `${diffMins} মিনিট আগে`; if (diffHours < 24) return `${diffHours} ঘণ্টা আগে`; return postDate.toLocaleDateString('bn-BD'); } // পেজ লোড হলে রান হবে document.addEventListener('DOMContentLoaded', fetchFeaturedNews);
